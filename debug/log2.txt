    local next_is_model=false

    # Parse arguments to see what model is being requested
    for arg in "$@"; do
        if [[ "$next_is_model" == true ]]; then
            target_model="$arg"
            next_is_model=false
        elif [[ "$arg" == "--model" ]]; then
            next_is_model=true
        fi
    done

    # Base environment variables
    export GITHUB_COPILOT_ENTERPRISE_URL=https://github.com/enterprises/cbcrc

    export COPILOT_MODEL="$target_model"

    # Route conditionally, masking VIRTUAL_ENV inline to suppress cross-env warnings
