# Headroom wrapped GitHub Copilot subscription CLI
export HEADROOM_DIR="$HOME/Dev/headroom"
export UV_SKIP_WHEEL_FILENAME_CHECK=1

copilot-h() {
  local target_model="gpt-4o"
  local args=("$@")
  local next_is_model=false

  for arg in "$@"; do
    if [[ "$next_is_model" == true ]]; then
      target_model="$arg"
      next_is_model=false
    elif [[ "$arg" == "--model" || "$arg" == "-m" ]]; then
      next_is_model=true
    elif [[ "$arg" == --model=* ]]; then
      target_model="${arg#--model=}"
    fi
  done

  # GitHub.com Enterprise Cloud is not a host override.
  # Headroom will discover https://api.business.githubcopilot.com itself.
  unset GITHUB_COPILOT_ENTERPRISE_URL
  unset GITHUB_COPILOT_ENTERPRISE_DOMAIN

  VIRTUAL_ENV="" uv run --project "$HEADROOM_DIR" --extra all --with fastapi --with uvicorn -- \
    headroom wrap copilot --subscription --debug-copilot -- "$@"
}

copilot-h-debug() {
  unset GITHUB_COPILOT_ENTERPRISE_URL
  unset GITHUB_COPILOT_ENTERPRISE_DOMAIN

  VIRTUAL_ENV="" uv run --project "$HEADROOM_DIR" --extra all --with fastapi --with uvicorn -- \
    python "$HEADROOM_DIR/scripts/debug_copilot_subscription.py" --model "${1:-gpt-4o}" --header-style all
}
